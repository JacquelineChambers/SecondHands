cmake_minimum_required(VERSION 3.24)
project(secondhands_qnn_stream LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(EXECUTORCH_BUILD_QNN ON CACHE BOOL "" FORCE)
set(EXECUTORCH_BUILD_XNNPACK OFF CACHE BOOL "" FORCE)
set(EXECUTORCH_BUILD_PTHREADPOOL OFF CACHE BOOL "" FORCE)
set(EXECUTORCH_BUILD_CPUINFO OFF CACHE BOOL "" FORCE)
set(EXECUTORCH_BUILD_EXECUTOR_RUNNER OFF CACHE BOOL "" FORCE)
set(EXECUTORCH_BUILD_EXTENSION_MODULE ON CACHE BOOL "" FORCE)
set(EXECUTORCH_BUILD_EXTENSION_DATA_LOADER ON CACHE BOOL "" FORCE)
set(EXECUTORCH_BUILD_EXTENSION_FLAT_TENSOR ON CACHE BOOL "" FORCE)
set(EXECUTORCH_BUILD_EXTENSION_NAMED_DATA_MAP ON CACHE BOOL "" FORCE)
set(EXECUTORCH_BUILD_EXTENSION_TENSOR ON CACHE BOOL "" FORCE)
set(EXECUTORCH_ENABLE_LOGGING ON CACHE BOOL "" FORCE)
set(EXECUTORCH_ENABLE_EVENT_TRACER OFF CACHE BOOL "" FORCE)
set(EXECUTORCH_BUILD_DEVTOOLS OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../executorch" executorch-runtime)
add_executable(secondhands_qnn_stream qnn_stream.cpp)
target_link_libraries(secondhands_qnn_stream PRIVATE
    extension_module_static extension_tensor executorch qnn_executorch_backend)
add_custom_command(TARGET secondhands_qnn_stream POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
    $<TARGET_FILE:qnn_executorch_backend> $<TARGET_FILE_DIR:secondhands_qnn_stream>)
